Flow

SIGTERM
  |
stop accepting traffic
  |
finish in-flight work
  |
close resources
  |
exit

Go

ctx, stop := signal.NotifyContext(
    context.Background(),
    syscall.SIGTERM,
    syscall.SIGINT,
)
defer stop()
<-ctx.Done()
